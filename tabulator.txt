func closestMatch(for value: String, from potentialMatches: [String]) -> String {
    var bestDistance = Int.max
    var bestIndex = 0

    for i in 0..<potentialMatches.count {
        let candidate = potentialMatches[i]
        let d = editDistance(from: value, to: candidate)

        if d < bestDistance {
            bestDistance = d
            bestIndex = i
        }
    }

    return potentialMatches[bestIndex]
}

var tabulator = Tabulator()

var lowercaseCatalog: [String] = []
for show in showCatalog {
    lowercaseCatalog.append(show.lowercased())
}

var errorCount = 0

for raw in surveyData {
    let cleaned = raw.lowercased()

    if lowercaseCatalog.contains(cleaned) {
        tabulator.incrementCount(forValue: cleaned)
    } else {
        let corrected = closestMatch(for: cleaned, from: lowercaseCatalog)
        tabulator.incrementCount(forValue: corrected)
        errorCount += 1
    }
}

for show in lowercaseCatalog {
    let count = tabulator.count(forValue: show)
    if count > 0 {
        print("\(show): \(count)")
    }
}

print("\n***** DATA ERRORS (FIXED) *****\n")
print("Total corrections made: \(errorCount)")
